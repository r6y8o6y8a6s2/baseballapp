import streamlit as st

# アプリのタイトルを設定
st.title("⚾ プロ野球・打率計算アプリ")

# サブタイトルや説明文
st.write("打数と安打数を入力すると、自動で打率を計算します。")

# 入力フォームを作成（数値入力）
ab = st.number_input("打数 (At Bats) を入力してください", min_value=1, value=100)
hits = st.number_input("安打数 (Hits) を入力してください", min_value=0, value=30)

# 計算ボタンが押されたときの処理
if st.button("打率を計算する"):
    if hits > ab:
        st.error("エラー：安打数が打数を超えています！")
    else:
        # 打率の計算（小数第3位まで表示）
        avg = hits / ab
        
        # 結果を表示
        st.success(f"結果：打率 **{avg:.3f}** （{avg:.3f}）")
        
        # 評価コメントのオマケ
        if avg >= 0.300:
            st.balloons()  # 画面に風船が飛ぶエフェクト！
            st.write("🎉 素晴らしい！3割打者（クリーンナップ級）です！")
        elif avg >= 0.250:
            st.write("👍 ナイスバッティング！レギュラー級の成績です！")
        else:
            st.write("💪 もっと練習してレギュラーを奪い返そう！")
